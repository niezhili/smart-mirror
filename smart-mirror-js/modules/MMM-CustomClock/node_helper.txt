const NodeHelper = require("node_helper");

module.exports = NodeHelper.create({
	start: function () {
		console.log("This is CustomClock start state:")
	},
	socketNotificationReceived: function (notification, payload) {
		if (notification === "CONFIG") {
			this.config = payload;
			this.checkLanguage();
		}
	},
	checkLanguage: function () {
//    	console.log("Language check in node_helper: ", this.config.language)
		if (this.config.language === "Zh-cn") {
			this.sendSocketNotification("HIDE_MODULE");
		} else {
			this.sendSocketNotification("SHOW_MODULE");
		}
	}

});
